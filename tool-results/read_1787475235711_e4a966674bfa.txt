   122→# ━━ TocDocTemplate ━━
   123→class TocDocTemplate(SimpleDocTemplate):
   124→    def afterFlowable(self, flowable):
   125→        if hasattr(flowable, 'bookmark_name'):
   126→            level = getattr(flowable, 'bookmark_level', 0)
   127→            text = getattr(flowable, 'bookmark_text', '')
   128→            key = getattr(flowable, 'bookmark_key', '')
   129→            self.notify('TOCEntry', (level, text, self.page, key))
   130→
   131→# ━━ Helpers ━━
   132→def heading(text, style, level=0):
   133→    key = 'h_%s' % hashlib.md5(text.encode()).hexdigest()[:8]
   134→    p = Paragraph('<a name="%s"/>%s' % (key, text), style)
   135→    p.bookmark_name = text
   136→    p.bookmark_level = level
   137→    p.bookmark_text = text
   138→    p.bookmark_key = key
   139→    return p

   141→def safe_keep(elements):
   142→    total = 0
   143→    for el in elements:
   144→        w, h = el.wrap(AVAIL_W, PAGE_H)
   145→        total += h
   146→    max_h = PAGE_H * 0.4
   147→    if total <= max_h:
   148→        return [KeepTogether(elements)]
   149→    elif len(elements) >= 2:
   150→        return [KeepTogether(elements[:2])] + list(elements[2:])
   151→    return list(elements)

   153→def make_table(headers, rows, col_ratios=None):
   154→    if col_ratios is None:
   155→        col_ratios = [1.0 / len(headers)] * len(headers)
   156→    col_widths = [r * AVAIL_W for r in col_ratios]
   157→    # Scale up if too narrow
   158→    tw = sum(col_widths)
   159→    if tw < AVAIL_W * 0.85:
   160→        scale = (AVAIL_W * 0.92) / tw
   161→        col_widths = [w * scale for w in col_widths]
   162→    data = [[Paragraph('<b>%s</b>' % h, th_style) for h in headers]]
   163→    for row in rows:
   164→        data.append([Paragraph(str(c), td_style) if not isinstance(c, Paragraph) else c for c in row])
   165→    t = Table(data, colWidths=col_widths, hAlign='CENTER')
   166→    style_cmds = [
   167→        ('BACKGROUND', (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   168→        ('TEXTCOLOR', (0, 0), (-1, 0), colors.white),
   169→        ('GRID', (0, 0), (-1, -1), 0.4, BORDER),
   170→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   171→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   172→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   173→        ('TOPPADDING', (0, 0), (-1, -1), 5),
   174→        ('BOTTOMPADDING', (0, 0), (-1, -1), 5),
   175→    ]
   176→    for i in range(1, len(data)):
   177→        bg = TABLE_ROW_ODD if i % 2 == 0 else TABLE_ROW_EVEN
   178→        style_cmds.append(('BACKGROUND', (0, i), (-1, i), bg))
   179→    t.setStyle(TableStyle(style_cmds))
   180→    return t
